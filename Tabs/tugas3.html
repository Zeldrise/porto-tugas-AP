Tugas 3 - Implementasi Algoritma Perulangan (Looping)

Tugas Pemrograman Awal:

Pada modul praktikum ini, tugas awal yang diberikan adalah memahami dan membuat struktur perulangan untuk mengeksekusi blok kode secara berulang selama kondisi terpenuhi. Jenis perulangan yang dipelajari meliputi:

  • Perulangan for: Digunakan saat jumlah perulangan sudah diketahui secara pasti (Determinate Loop).
  • Perulangan while: Digunakan ketika jumlah perulangan belum diketahui pasti dan bergantung pada kondisi tertentu (Indeterminate Loop).
  • Struktur Bersarang (Nested Loop): Perulangan di dalam perulangan untuk menangani kasus kompleks.

Sebagai studi kasus utama, algoritma perulangan ini diimplementasikan ke dalam Sistem Kasir Swalayan Sederhana untuk mendata banyak jenis barang belanjaan yang dibeli pelanggan secara dinamis.

Source Code JavaScript

Berikut adalah kode program asli Node.js berbasis async/await yang dijalankan menggunakan VS Code terminal:

const readline = require('readline/promises');
const { stdin: input, stdout: output } = require('process');

async function main() {
    const rl = readline.createInterface({ input, output });
    let pilihanMenu;

    do {
        console.clear();
        console.log("========================================");
        console.log("    SISTEM KASIR SWALAYAN SEDERHANA     ");
        console.log("========================================");
        console.log("1. Mulai Transaksi Belanja");
        console.log("2. Keluar dari Program\n");
        
        pilihanMenu = await rl.question("Pilih menu (1/2): ");

        if (pilihanMenu === "1") {
            let inputBarang = await rl.question("\nMasukkan berapa banyak jenis barang yang dibeli: ");
            let jumlahJenisBarang = parseInt(inputBarang);
            
            if (isNaN(jumlahJenisBarang) || jumlahJenisBarang <= 0) {
                console.log("Jumlah barang tidak valid!");
                await rl.question("\nTekan Enter untuk melanjutkan...");
                continue; 
            }

            // 2. STRUKTUR FOR (PERULANGAN BARANG)
            for (let i = 1; i <= jumlahJenisBarang; i++) {
                console.log(`\n--- Data Barang ke-${i} ---`);
                
                let namaBarang = await rl.question(`[Barang ke-${i}] Masukkan Nama Barang: `);
                let inputHarga = await rl.question(`[Barang ke-${i}] Masukkan Harga Satuan: `);
                let inputJumlah = await rl.question(`[Barang ke-${i}] Masukkan Jumlah Beli: `);

                let hargaBarang = parseInt(inputHarga);
                let jumlahBeli = parseInt(inputJumlah);

                let totalPerBarang = hargaBarang * jumlahBeli;
                console.log(`Nama Barang  : ${namaBarang}`);
                console.log(`Harga Satuan : Rp ${hargaBarang}`);
                console.log(`Jumlah Beli  : ${jumlahBeli}`);
                console.log(`Total Harga  : Rp ${totalPerBarang}`);

                // 3. PERULANGAN BERSARANG (NESTED LOOP GRAFIK BINTANG)
                let grafikBintang = "";
                for (let j = 1; j <= jumlahBeli; j++) {
                    grafikBintang += "*";
                }
                console.log(`Grafik Jumlah: ${grafikBintang} (${jumlahBeli} pcs)`);
            }

            // 4. STRUKTUR WHILE (VALIDASI LUNAS)
            let konfirmasi = "";
            while (konfirmasi.toUpperCase() !== "LUNAS") {
                konfirmasi = await rl.question("\n[VALIDASI] Ketik 'LUNAS' untuk menyelesaikan pembayaran: ");
                
                if (konfirmasi.toUpperCase() !== "LUNAS") {
                    console.log("Status salah! Pembayaran belum diterima.");
                }
            }
            
            console.log("\nTerima kasih! Pembayaran berhasil diverifikasi.");
            console.log("Status Pembayaran: LUNAS");

        } else if (pilihanMenu === "2") {
            console.log("Sampai jumpa kembali!");
        } else {
            console.log("Pilihan tidak valid! Silakan coba lagi.");
            await rl.question("\nTekan Enter untuk melanjutkan...");
        }

        if (pilihanMenu !== "2") {
            let konfirmasiMenu = await rl.question("\nKembali ke menu utama? (y/n): ");
            if (konfirmasiMenu.toLowerCase() === "n") {
                pilihanMenu = "2"; 
                console.log("Program selesai. Terima kasih!");
            }
        }

    } while (pilihanMenu !== "2");

    rl.close();
}
main();
